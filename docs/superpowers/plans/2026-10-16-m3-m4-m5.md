import { applyPresence, presenceOf } from '../src/client/workgroup-presence.js'

// A workgroup's member list has to answer "is this colleague at their desk" for the people in the
// room — and nothing else. The Hub reports presence as a boolean per member, derived from live
// connections, never from `last_seen_at`; a client that invented its own notion of "recently
// active" would disagree with the Hub about who is in the room.

test('a member the Hub reported as connected reads as online', () => {
  assert.equal(presenceOf({ userId: 'u1', online: true }), 'online')
})

test('a member without the flag, or with it false, reads as offline', () => {
  // The Hub omits `online` when it has nothing to say. Absent must not read as online: a member
  // list that assumed the best would keep showing colleagues who already left.
  assert.equal(presenceOf({ userId: 'u1', online: false }), 'offline')
  assert.equal(presenceOf({ userId: 'u1' }), 'offline')
  assert.equal(presenceOf(undefined), 'offline')
})

test('a presence event changes exactly one member and leaves the rest alone', () => {
  const members = [
    { userId: 'u1', online: true },
    { userId: 'u2', online: true },
  ]
  const next = applyPresence(members, { userId: 'u1', online: false })
  assert.deepEqual(next.map(m => m.online), [false, true])
  // A new array, so React sees a change; the untouched row keeps its identity, so only the row
  // whose state changed re-renders.
  assert.notEqual(next, members)
  assert.equal(next[1], members[1])
})

test('an event for someone not in this room is ignored', () => {
  // Presence is broadcast per room, but a member may have been removed between the event being
  // queued and being applied. Adding them back from an event would resurrect a removed member.
  const members = [{ userId: 'u1', online: false }]
  const next = applyPresence(members, { userId: 'gone', online: true })
  assert.equal(next.length, 1)
  assert.deepEqual(next.map(m => m.online), [false])
})

  // The event carries only a boolean. A client that copied the whole payload into the member row
  // would let a server-chosen field shadow the row the Hub actually sent.
  const members = [{ userId: 'u1', userHandle: 'ravi', online: false }]
  const next = applyPresence(members, { userId: 'u1', online: true, displayName: 'not-the-real-name' })
  assert.deepEqual(next[0], { userId: 'u1', userHandle: 'ravi', online: true })

```js
// src/client/workgroup-presence.js
// Who is at their desk, as the Hub sees it.
//
// Presence is a property of the *connection*, not of the person and not of the device: the Hub
// reports a member online while any of their devices is connected, and offline once none is. So the
// only correct client behaviour is to render the boolean the Hub sends and to apply the events that
// change it — never to infer freshness from a timestamp, and never to keep a member online because
// they were a moment ago.
//
// Zero dependencies, so the rules can be read and checked without a DOM.

/** The two states a member can be in, as the member list shows them. */
export const PRESENCE_STATES = ['online', 'offline']

/**
 * Whether one member is connected.
 *
 * A member row without the flag is offline: the Hub omits it when it has nothing to say, and
 * assuming online would leave departed colleagues in the list.
 *
 * @param member - a member row from the Hub, possibly carrying `online`.
 * @returns `'online'` or `'offline'`.
 */
export function presenceOf(member) {
  return member?.online === true ? 'online' : 'offline'
}

/**
 * Apply one `presence.changed` event to the member list.
 *
 * @param members - the current member rows.
 * @param event - `{ userId, online }`, as the Bridge delivers it.
 * @returns a new array when the named member is present, otherwise the input unchanged; the rows
 *   that were not named keep their identity so only the changed row re-renders.
 */
export function applyPresence(members, event) {
  const list = members ?? []
  const index = list.findIndex(member => member.userId === event?.userId)
  // An event for someone this room does not list is dropped rather than appended: a member removed
  // a moment ago must not reappear because a broadcast about them was still in flight.
  if (index < 0) return list
  const next = [...list]
  next[index] = { ...list[index], online: event.online === true }
  return next
}
```

import { test } from 'node:test'
import { judgeEgress } from '../src/security/egress.js'

// Outbound access is decided by an explicit allow-list and defaults to refused. The reverse —
// allowing everything and blocking known-bad hosts — cannot be made safe, because the set of hosts
// that should be reachable is finite and known while the set that should not is neither.
test('a host outside the approved list is refused', () => {
  const verdict = judgeEgress({ host: 'example.com', port: 443 }, { allowed: ['internal.corp'] })
  assert.equal(verdict.allowed, false)
  assert.match(verdict.reason, /not approved/i)
})

test('an empty list refuses everything rather than allowing everything', () => {
  // The dangerous default would be "no policy means no restriction". A task with no approved egress
  // has no business reaching the network at all.
  assert.equal(judgeEgress({ host: 'example.com', port: 443 }, { allowed: [] }).allowed, false)
  assert.equal(judgeEgress({ host: 'example.com', port: 443 }, {}).allowed, false)
})

test('an approved host is allowed, with or without an explicit port', () => {
  assert.equal(judgeEgress({ host: 'internal.corp', port: 443 }, { allowed: ['internal.corp'] }).allowed, true)
  assert.equal(judgeEgress({ host: 'internal.corp', port: 8080 }, { allowed: ['internal.corp'] }).allowed, true)
})

test('a port-restricted approval does not widen to other ports', () => {
  const policy = { allowed: ['internal.corp:8443'] }
  assert.equal(judgeEgress({ host: 'internal.corp', port: 8443 }, policy).allowed, true)
  assert.equal(judgeEgress({ host: 'internal.corp', port: 22 }, policy).allowed, false)
})

test('a lookalike domain is refused', () => {
  // The classic bypass: `internal.corp.attacker.example` ends with the approved string.
  assert.equal(judgeEgress({ host: 'internal.corp.attacker.example', port: 443 }, { allowed: ['internal.corp'] }).allowed, false)
})

test('case is not a way around the list', () => {
  assert.equal(judgeEgress({ host: 'Internal.Corp', port: 443 }, { allowed: ['internal.corp'] }).allowed, true)
})
